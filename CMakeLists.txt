cmake_minimum_required(VERSION 3.16)
project(saber_rider C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()
find_package(SDL3 REQUIRED CONFIG)
file(GLOB SRC ${CMAKE_SOURCE_DIR}/src/*.c)
add_executable(saber_rider ${SRC})
target_compile_options(saber_rider PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(saber_rider PRIVATE SDL3::SDL3 m)
